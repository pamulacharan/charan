package org.example;

import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.io.PrintWriter;
import java.sql.CallableStatement;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

@WebServlet("/DeleteServlet")
public class DeleteServlet extends HttpServlet {
    private static final long serialVersionUID = 1L;

    private static final Logger LOGGER = Logger.getLogger(DeleteServlet.class.getName());

    // Same database credentials as your LoginServlet
    private static final String DB_URL = "jdbc:oracle:thin:@//SMTRND:1531/SMTRND";
    private static final String DB_USER = "Linus";
    private static final String DB_PASSWORD = "G4uh3yAW!";

    // Stored procedure call - exactly like your request
    private static final String SQL_CALL =
            "{ call linus.pkg_claim_document_deletion.process_invoice_deletion(?, ?) }";

    // Explicitly load the Oracle driver – same as LoginServlet
    static {
        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");
        } catch (ClassNotFoundException e) {
            throw new RuntimeException("Oracle JDBC driver not found. " +
                    "Please place ojdbc8.jar in WEB-INF/lib.", e);
        }
    }

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        doPost(request, response);
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        response.setContentType("text/plain");
        PrintWriter out = response.getWriter();

        String docIdStr = request.getParameter("docId");
        String incidentNo = request.getParameter("incidentNo");

        // Input validation – same pattern as LoginServlet
        if (docIdStr == null || docIdStr.trim().isEmpty() ||
                incidentNo == null || incidentNo.trim().isEmpty()) {
            LOGGER.warning("Deletion attempt with empty parameters");
            out.print("Error: Both Document ID and Incident Number are required.");
            return;
        }

        long docId;
        try {
            docId = Long.parseLong(docIdStr.trim());
        } catch (NumberFormatException e) {
            LOGGER.warning("Invalid document ID: " + docIdStr);
            out.print("Error: Document ID must be a valid number.");
            return;
        }

        // Execute stored procedure – same try-with-resources pattern as LoginServlet
        try (Connection conn = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             CallableStatement cstmt = conn.prepareCall(SQL_CALL)) {

            cstmt.setLong(1, docId);
            cstmt.setString(2, incidentNo);
            cstmt.execute();

            LOGGER.info("Document deleted successfully: " + docId + " (incident: " + incidentNo + ")");
            out.print("success: Document " + docId + " deleted successfully (incident: " + incidentNo + ")");

        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE, "Database error during deletion for document: " + docId, e);
            out.print("Database error: " + e.getMessage());
        }
    }
}
