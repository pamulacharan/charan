## 📤 4. Servlet: `SubmitRequestServlet`

Handles the user’s submission – inserts a new row into `deletion_requests`.

```java
package org.example;

import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import javax.servlet.http.HttpSession;
import java.io.IOException;
import java.sql.*;
import java.util.logging.Level;
import java.util.logging.Logger;

@WebServlet("/SubmitRequestServlet")
public class SubmitRequestServlet extends HttpServlet {
    private static final long serialVersionUID = 1L;
    private static final Logger LOGGER = Logger.getLogger(SubmitRequestServlet.class.getName());

    private static final String DB_URL = "jdbc:oracle:thin:@//SMTRND:1531/SMTRND";
    private static final String DB_USER = "Linus";
    private static final String DB_PASSWORD = "G4uh3yAW!";

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        HttpSession session = request.getSession();
        String username = (String) session.getAttribute("username");
        if (username == null) {
            response.sendRedirect("login.jsp");
            return;
        }

        String docIdStr = request.getParameter("docId");
        String incidentNo = request.getParameter("incidentNo");

        if (docIdStr == null || docIdStr.trim().isEmpty() ||
            incidentNo == null || incidentNo.trim().isEmpty()) {
            request.setAttribute("message", "Both fields are required.");
            request.setAttribute("type", "error");
            request.getRequestDispatcher("user.jsp").forward(request, response);
            return;
        }

        long docId;
        try {
            docId = Long.parseLong(docIdStr.trim());
        } catch (NumberFormatException e) {
            request.setAttribute("message", "Document ID must be a number.");
            request.setAttribute("type", "error");
            request.getRequestDispatcher("user.jsp").forward(request, response);
            return;
        }

        // Insert into deletion_requests
        String insertSql = "INSERT INTO deletion_requests (request_id, doc_id, incident_no, submitted_by, status) " +
                           "VALUES (deletion_requests_seq.NEXTVAL, ?, ?, ?, 'PENDING')";

        try (Connection conn = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             PreparedStatement pst = conn.prepareStatement(insertSql)) {

            pst.setLong(1, docId);
            pst.setString(2, incidentNo);
            pst.setString(3, username);
            int rows = pst.executeUpdate();

            if (rows > 0) {
                request.setAttribute("message", "Request submitted successfully!");
                request.setAttribute("type", "success");
            } else {
                request.setAttribute("message", "Submission failed.");
                request.setAttribute("type", "error");
            }
            LOGGER.info("User " + username + " submitted deletion request for doc " + docId);

        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE, "Insert error", e);
            request.setAttribute("message", "Database error: " + e.getMessage());
            request.setAttribute("type", "error");
        }

        request.getRequestDispatcher("user.jsp").forward(request, response);
    }
}
```

## 🗑️ 6. Servlet: `AdminProcessServlet`

This servlet:
- Receives a `requestId`
- Fetches `doc_id` and `incident_no` from the table
- Calls the stored procedure
- Updates the request status to `DELETED` (or `FAILED`) and records the admin

```java
package org.example;

import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import javax.servlet.http.HttpSession;
import java.io.IOException;
import java.sql.*;
import java.util.logging.Level;
import java.util.logging.Logger;

@WebServlet("/AdminProcessServlet")
public class AdminProcessServlet extends HttpServlet {
    private static final long serialVersionUID = 1L;
    private static final Logger LOGGER = Logger.getLogger(AdminProcessServlet.class.getName());

    private static final String DB_URL = "jdbc:oracle:thin:@//SMTRND:1531/SMTRND";
    private static final String DB_USER = "Linus";
    private static final String DB_PASSWORD = "G4uh3yAW!";

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        HttpSession session = request.getSession();
        String admin = (String) session.getAttribute("username");
        if (admin == null || !"ADMIN".equalsIgnoreCase((String) session.getAttribute("role"))) {
            response.sendRedirect("login.jsp");
            return;
        }

        String requestIdStr = request.getParameter("requestId");
        if (requestIdStr == null || requestIdStr.trim().isEmpty()) {
            request.setAttribute("message", "Missing request ID.");
            request.setAttribute("type", "error");
            request.getRequestDispatcher("admin.jsp").forward(request, response);
            return;
        }

        int requestId;
        try {
            requestId = Integer.parseInt(requestIdStr.trim());
        } catch (NumberFormatException e) {
            request.setAttribute("message", "Invalid request ID.");
            request.setAttribute("type", "error");
            request.getRequestDispatcher("admin.jsp").forward(request, response);
            return;
        }

        // Fetch doc_id and incident_no for this request
        String selectSql = "SELECT doc_id, incident_no FROM deletion_requests WHERE request_id = ? AND status = 'PENDING'";
        long docId = 0;
        String incidentNo = null;

        try (Connection conn = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             PreparedStatement pst = conn.prepareStatement(selectSql)) {
            pst.setInt(1, requestId);
            try (ResultSet rs = pst.executeQuery()) {
                if (rs.next()) {
                    docId = rs.getLong("doc_id");
                    incidentNo = rs.getString("incident_no");
                } else {
                    request.setAttribute("message", "Request not found or already processed.");
                    request.setAttribute("type", "error");
                    request.getRequestDispatcher("admin.jsp").forward(request, response);
                    return;
                }
            }
        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE, "Fetch error", e);
            request.setAttribute("message", "DB error: " + e.getMessage());
            request.setAttribute("type", "error");
            request.getRequestDispatcher("admin.jsp").forward(request, response);
            return;
        }

        // Now call the stored procedure
        String procCall = "{ call linus.pkg_claim_document_deletion.process_invoice_deletion(?, ?) }";
        boolean success = false;
        String statusUpdate = "FAILED";
        try (Connection conn = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             CallableStatement cstmt = conn.prepareCall(procCall)) {

            cstmt.setLong(1, docId);
            cstmt.setString(2, incidentNo);
            cstmt.execute();
            success = true;
            statusUpdate = "DELETED";
            LOGGER.info("Admin " + admin + " deleted doc " + docId + " (request " + requestId + ")");

        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE, "Procedure execution failed", e);
            // still update status to FAILED
        }

        // Update the request status
        String updateSql = "UPDATE deletion_requests SET status = ?, processed_by = ?, processed_date = CURRENT_TIMESTAMP " +
                           "WHERE request_id = ?";
        try (Connection conn = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             PreparedStatement pst = conn.prepareStatement(updateSql)) {
            pst.setString(1, statusUpdate);
            pst.setString(2, admin);
            pst.setInt(3, requestId);
            pst.executeUpdate();
        } catch (SQLException e) {
            LOGGER.log(Level.SEVERE, "Update status failed", e);
        }

        // Set message
        if (success) {
            request.setAttribute("message", "Document " + docId + " deleted successfully.");
            request.setAttribute("type", "success");
        } else {
            request.setAttribute("message", "Deletion failed. Check logs for details.");
            request.setAttribute("type", "error");
        }
        request.getRequestDispatcher("admin.jsp").forward(request, response);
    }
}
```

## 📦 Deployment Overview

1. **Run the SQL scripts** to create the table and update the role column.
2. **Place all JSP files** (`login.jsp`, `user.jsp`, `admin.jsp`) in your web root.
3. **Compile all servlets** (`LoginServlet`, `SubmitRequestServlet`, `AdminProcessServlet`, `LogoutServlet`) – all in package `org.example`.
4. **Ensure `ojdbc8.jar` is in `WEB-INF/lib`**.
5. **Restart Tomcat**.
6. **Access** `http://localhost:8080/yourapp/login.jsp`.

## ✅ Flow Summary

1. **User logs in** – role is stored in session.
2. **If user** → redirected to `user.jsp` → can submit requests → stored in DB with `PENDING`.
3. **If admin** → redirected to `admin.jsp` → sees pending requests → clicks **Delete** → the servlet calls the stored procedure, updates status to `DELETED` or `FAILED`.
4. **Both can view history** (users see only their own, admins see all).

All code is **Java 1.8 compatible** – no lambdas, no modules, pure JDBC + Servlets + JSP.
