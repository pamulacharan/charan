import java.io.IOException;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;

import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

@WebServlet("/Login")
public class Login extends HttpServlet {
    private static final long serialVersionUID = 1L;

    // Database credentials – MOVE THESE TO A PROPERTIES FILE OR ENVIRONMENT VARIABLES!
    private static final String DB_URL = "jdbc:oracle:thin:@SMTRND:1531:SMTRND";
    private static final String DB_USER = "Linus";
    private static final String DB_PASSWORD = "G4uh3yAW!";

    // SQL query
    private static final String SELECT_USER_SQL = "SELECT * FROM sv_user WHERE USER_ID=? AND PASSWORD=?";

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        String username = request.getParameter("textName");
        String password = request.getParameter("textpwd");

        // Basic input validation
        if (username == null || username.trim().isEmpty() || password == null || password.trim().isEmpty()) {
            response.sendRedirect("Error.jsp");
            return;
        }

        try (Connection conn = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             PreparedStatement pst = conn.prepareStatement(SELECT_USER_SQL)) {

            pst.setString(1, username);
            pst.setString(2, password);

            try (ResultSet rs = pst.executeQuery()) {
                if (rs.next()) {
                    response.sendRedirect("Succes.jsp");
                } else {
                    response.sendRedirect("Error.jsp");
                }
            }

        } catch (SQLException e) {
            // Log the exception (use a logging framework in production)
            log("Authentication failed for user: " + username, e);
            response.sendRedirect("Error.jsp");
        }
    }
}
