package com.pamula;

import java.io.IOException;
import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.logging.Level;
import java.util.logging.Logger;

import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

import com.pamula.util.DBConnection;

@WebServlet("/LoginServlet")
public class LoginServlet extends HttpServlet {
    private static final long serialVersionUID = 1L;

    // Use java.util.logging (or switch to SLF4J if you have it)
    private static final Logger LOGGER = Logger.getLogger(LoginServlet.class.getName());

    private static final String SQL_SELECT = "SELECT * FROM SV_USER WHERE USER_ID = ? AND PASSWORD = ?";

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        // Optionally forward to a login page or simply call doPost
        doPost(request, response);
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        String username = request.getParameter("textName");
        String password = request.getParameter("textpwd");

        // 1. Input validation
        if (username == null || username.trim().isEmpty() ||
            password == null || password.trim().isEmpty()) {
            LOGGER.warning("Login attempt with empty credentials");
            response.sendRedirect(request.getContextPath() + "/Error.jsp");
            return;
        }

        // 2. Authenticate user
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement pst = conn.prepareStatement(SQL_SELECT)) {

            pst.setString(1, username);
            pst.setString(2, password);

            try (ResultSet rs = pst.executeQuery()) {
                if (rs.next()) {
                    LOGGER.info("User logged in successfully: " + username);
                    response.sendRedirect(request.getContextPath() + "/Succes.jsp");
                } else {
                    LOGGER.info("Failed login attempt for user: " + username);
                    response.sendRedirect(request.getContextPath() + "/Error.jsp");
                }
            }

        } catch (SQLException e) {
            // Log the full error for debugging, but show a generic error page to the user
            LOGGER.log(Level.SEVERE, "Database error during login for user: " + username, e);
            response.sendRedirect(request.getContextPath() + "/Error.jsp");
        }
    }
}
