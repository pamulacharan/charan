FUNCTION validate_invoice(p_document_id IN NUMBER)
RETURN BOOLEAN
IS
    v_count NUMBER;
BEGIN
    SELECT COUNT(*)
      INTO v_count
      FROM sv_invoice_header
     WHERE document_id   = p_document_id
       AND posting_status = 0
       AND error_id       IS NULL
       AND profit_center_id = 1054;

    RETURN v_count > 0;

EXCEPTION
    WHEN OTHERS THEN
        RETURN FALSE;
END validate_invoice;

PROCEDURE process_invoice_deletion(
    p_document_id IN NUMBER,
    p_ticket      IN VARCHAR2
) IS
BEGIN
    -- Step 1: Validate invoice using existing function
      IF NOT validate_invoice(p_document_id) THEN
        log_event(
            p_doc_id  => p_document_id,
            p_ticket  => p_ticket,
            p_action  => 'VALIDATION_FAILED',
            p_status  => 'FAILED',
            p_message => 'Invoice failed validation and cannot be deleted. Checked on ' || SYSDATE
        );
        RETURN;
    END IF;

    -- Step 2: If valid, call the original procedure
     backup_and_purge_invoice(p_document_id, p_ticket);

EXCEPTION
    WHEN OTHERS THEN
        log_event(
            p_doc_id  => p_document_id,
            p_ticket  => p_ticket,
            p_action  => 'PROCESS_INVOICE_DELETION',
            p_status  => 'FAILED',
            p_message => 'Unexpected error in wrapper: ' || SQLERRM
        );
        RAISE;
END process_invoice_deletion;

END pkg_claim_document_deletion;
