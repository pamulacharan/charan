        <form id="deleteForm" action="<%= request.getContextPath() %>/DeleteDocumentServlet" method="post">
            <div class="form-group">
                <label>Document ID</label>
                <input type="number" name="docId" required placeholder="23409461">
            </div>
            <div class="form-group">
                <label>Incident Number</label>
                <input type="text" name="incidentNo" required placeholder="INC1925519">
            </div>
            <input type="submit" value="Execute">
        </form>
        <div id="message" class="message"></div>
    </div>

    <script>
        document.getElementById('deleteForm').addEventListener('submit', function(e) {
            e.preventDefault();
            var form = this;
            var data = new FormData(form);
            var xhr = new XMLHttpRequest();
            xhr.open('POST', form.action, true);
            xhr.onload = function() {
                var msg = document.getElementById('message');
                msg.style.display = 'block';
                if (xhr.responseText.indexOf('success') !== -1) {
                    msg.className = 'message success';
                } else {
                    msg.className = 'message error';
                }
                msg.textContent = xhr.responseText;
            };
            xhr.onerror = function() {
                var msg = document.getElementById('message');
                msg.style.display = 'block';
                msg.className = 'message error';
                msg.textContent = 'Network error – check console.';
            };
            xhr.send(data);
        });
    </script>
